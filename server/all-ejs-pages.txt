<script>
    const socket = io();
    const interviewId = "<%= interview._id %>";

    socket.on("connect", () => {
        socket.emit("join-interview", {
            interviewId,
            role: "candidate"
        });
    });

    const enableMediaBtn = document.getElementById("enableMediaBtn");
    const localVideo = document.getElementById("localVideo");
    const remoteVideo = document.getElementById("remoteVideo");
    const mediaStatus = document.getElementById("mediaStatus");

    let candidateStream = null;
    let peerConnection = null;
    let pendingOffer = null;
    const pendingIceCandidates = [];

    const createPeerConnection = () => {
        if (peerConnection) {
            return;
        }

        peerConnection = new RTCPeerConnection();

        peerConnection.addEventListener("track", (event) => {
            if (event.streams[0]) {
                remoteVideo.srcObject = event.streams[0];
            }
        });

        peerConnection.addEventListener("icecandidate", (event) => {
            if (event.candidate) {
                socket.emit("webrtc-ice-candidate", {
                    interviewId,
                    candidate: event.candidate
                });
            }
        });
    };

    const answerPendingOffer = async () => {
        if (!candidateStream || !pendingOffer) {
            return;
        }

        createPeerConnection();

        try {
            await peerConnection.setRemoteDescription(pendingOffer);
            pendingOffer = null;

            candidateStream.getTracks().forEach((track) => {
                peerConnection.addTrack(track, candidateStream);
            });

            while (pendingIceCandidates.length > 0) {
                await peerConnection.addIceCandidate(
                    pendingIceCandidates.shift()
                );
            }

            const answer = await peerConnection.createAnswer();
            await peerConnection.setLocalDescription(answer);

            socket.emit("webrtc-answer", {
                interviewId,
                answer: peerConnection.localDescription
            });
        } catch (error) {
            console.error("Failed to answer interviewer offer:", error);
            mediaStatus.textContent = "Could not connect interview media.";
        }
    };

    enableMediaBtn.addEventListener("click", async () => {
        try {
            candidateStream = await navigator.mediaDevices.getUserMedia({
                video: true,
                audio: true
            });

            localVideo.srcObject = candidateStream;
            createPeerConnection();
            enableMediaBtn.disabled = true;
            mediaStatus.textContent = "Camera and microphone enabled.";
            socket.emit("candidate-media-enabled", interviewId);

            await answerPendingOffer();
        } catch (error) {
            console.error("Candidate camera or microphone error:", error);
            mediaStatus.textContent =
                "Could not access camera or microphone.";
        }
    });

    socket.on("webrtc-offer", async (data) => {
        pendingOffer = data.offer;
        await answerPendingOffer();
    });

    socket.on("webrtc-ice-candidate", async (data) => {
        if (!peerConnection || !peerConnection.remoteDescription) {
            pendingIceCandidates.push(data.candidate);
            return;
        }

        try {
            await peerConnection.addIceCandidate(data.candidate);
        } catch (error) {
            console.error("Failed to add interviewer ICE candidate:", error);
        }
    });

    const startRecordingBtn = document.getElementById("startRecordingBtn");
    const stopRecordingBtn = document.getElementById("stopRecordingBtn");
    const recordingStatus = document.getElementById("recordingStatus");
    const transcriptResult = document.getElementById("transcriptResult");

    let mediaRecorder;
    let recordingStream = null;
    let audioChunks = [];

    startRecordingBtn.addEventListener("click", async () => {
        try {
            recordingStream = await navigator.mediaDevices.getUserMedia({
                audio: true
            });

            const options = MediaRecorder.isTypeSupported("audio/webm;codecs=opus")
                ? { mimeType: "audio/webm;codecs=opus" }
                : {};

            mediaRecorder = new MediaRecorder(recordingStream, options);
            audioChunks = [];

            console.log("Recording format:", mediaRecorder.mimeType);

            mediaRecorder.addEventListener("dataavailable", (event) => {
                if (event.data.size > 0) {
                    audioChunks.push(event.data);
                }
            });

            mediaRecorder.addEventListener("stop", async () => {
                try {
                    recordingStatus.textContent = "Sending audio for transcription...";

                    const audioBlob = new Blob(audioChunks, {
                        type: mediaRecorder.mimeType
                    });
                    console.log("Audio chunks:", audioChunks.length);
console.log("Audio blob size:", audioBlob.size);
console.log("Audio blob type:", audioBlob.type);
                    if (audioBlob.size === 0) {
                        throw new Error("Recording did not contain any audio data");
                    }

                    const formData = new FormData();
                    formData.append("audio", audioBlob, "recording.webm");

                    const response = await fetch("/transcribe-audio", {
                        method: "POST",
                        body: formData
                    });
                    const result = await response.json();

                    if (!response.ok) {
                        throw new Error(result.error || "Transcription failed");
                    }

                    console.log("Transcript received:", result.transcript);
                    transcriptResult.textContent = result.transcript;

                    socket.emit("transcript", {
                        interviewId: interviewId,
                        text: result.transcript,
                        speaker: "candidate"
                    });

                    recordingStatus.textContent = "Transcription completed.";
                } catch (error) {
                    console.error("Transcription error:", error);
                    recordingStatus.textContent = "Transcription failed.";
                } finally {
                    if (recordingStream) {
                        recordingStream.getTracks().forEach((track) => track.stop());
                        recordingStream = null;
                    }

                    startRecordingBtn.disabled = false;
                    stopRecordingBtn.disabled = true;
                }
            });

            mediaRecorder.start();
            recordingStatus.textContent = "Recording...";
            startRecordingBtn.disabled = true;
            stopRecordingBtn.disabled = false;
        } catch (error) {
            console.error("Microphone error:", error);
            recordingStatus.textContent = "Could not access microphone.";

            if (recordingStream) {
                recordingStream.getTracks().forEach((track) => track.stop());
                recordingStream = null;
            }
        }
    });

    stopRecordingBtn.addEventListener("click", () => {
        if (mediaRecorder && mediaRecorder.state !== "inactive") {
            mediaRecorder.stop();
            recordingStatus.textContent = "Processing audio...";
            stopRecordingBtn.disabled = true;
        }
    });

</script>

</body>
</html>

        /*
         * Get the scheduled interview time
         * from the Interview document.
         */
        const scheduledTime = new Date(
            "<%= interview.scheduledAt.toISOString() %>"
        ).getTime();


        /*
         * Get HTML elements.
         */
        const countdownTitle =
            document.getElementById("countdownTitle");

        const hoursElement =
            document.getElementById("hours");

        const minutesElement =
            document.getElementById("minutes");

        const secondsElement =
            document.getElementById("seconds");

        const statusMessage =
            document.getElementById("statusMessage");

        const consentCheckbox =
            document.getElementById("consentCheckbox");

        const startInterviewBtn =
            document.getElementById("startInterviewBtn");

        const waitingMessage =
            document.getElementById("waitingMessage");


        /*
         * Indicates whether the scheduled
         * interview time has arrived.
         */
        let interviewTimeReached = false;


        /*
         * Enable/disable Start Interview button.
         */
        function updateStartButton() {

    if (!startInterviewBtn || !consentCheckbox) {
        return;
    }

            startInterviewBtn.disabled = !(
                interviewTimeReached && consentCheckbox.checked
            );

            if (!interviewTimeReached) {
                waitingMessage.textContent =
                    "Please wait until the scheduled interview time.";
            } else if (!consentCheckbox.checked) {
                waitingMessage.textContent =
                    "Please confirm candidate consent to continue.";
            } else {
                waitingMessage.textContent =
                    "The interview is ready to start.";
            }
        }


        /*
         * Update countdown.
         */
        function updateCountdown() {

            const currentTime = Date.now();

            const difference =
                scheduledTime - currentTime;


            /*
             * Interview time has arrived.
             */
            if (difference <= 0) {

                interviewTimeReached = true;

                countdownTitle.textContent =
                    "Interview is ready to start";

                hoursElement.textContent = "00";
                minutesElement.textContent = "00";
                secondsElement.textContent = "00";

                statusMessage.textContent =
                    "The scheduled interview time has arrived.";

                updateStartButton();

                return;
            }


            /*
             * Convert remaining milliseconds
             * into hours, minutes and seconds.
             */
            const totalSeconds =
                Math.floor(difference / 1000);

            const hours =
                Math.floor(totalSeconds / 3600);

            const minutes =
                Math.floor(
                    (totalSeconds % 3600) / 60
                );

            const seconds =
                totalSeconds % 60;


            /*
             * Display countdown.
             */
            hoursElement.textContent =
                String(hours).padStart(2, "0");

            minutesElement.textContent =
                String(minutes).padStart(2, "0");

            secondsElement.textContent =
                String(seconds).padStart(2, "0");

            updateStartButton();
        }


        /*
         * When consent changes,
         * check whether Start Interview
         * can be enabled.
         */
        consentCheckbox.addEventListener(
            "change",
            updateStartButton
        );


        /*
         * Run once immediately.
         */
        updateCountdown();


        /*
         * Update countdown every second.
         */
        if (!interviewTimeReached) {
            const countdownTimer = setInterval(
                () => {
                    updateCountdown();

                    if (interviewTimeReached) {
                        clearInterval(countdownTimer);
                    }
                },
                1000
            );
        }
